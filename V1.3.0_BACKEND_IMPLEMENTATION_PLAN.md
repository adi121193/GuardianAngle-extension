function updateStatusChip(status) {
  const chip = document.getElementById('nerStatusChip');
  const label = chip.querySelector('.status-label');
  const icon = chip.querySelector('.status-icon');

  // Remove all status classes
  chip.className = 'status-chip';

  switch (status) {
    case 'disabled':
      chip.classList.add('status-disabled');
      label.textContent = 'NER: Disabled';
      icon.innerHTML = `<svg>...</svg>`; // X icon
      break;

    case 'loading':
      chip.classList.add('status-loading');
      label.textContent = 'NER: Loading...';
      icon.innerHTML = `<svg>...</svg>`; // Spinner icon
      break;

    case 'ready':
      chip.classList.add('status-ready');
      label.textContent = 'NER: Ready';
      icon.innerHTML = `<svg>...</svg>`; // Check icon
      break;

    case 'error':
      chip.classList.add('status-error');
      label.textContent = 'NER: Error';
      icon.innerHTML = `<svg>...</svg>`; // Warning icon
      break;
  }

  chip.setAttribute('aria-live', 'polite');
}

#### 5. Model Download Function
```javascript
async function startNERModelDownload() {
  const progressBar = document.getElementById('nerProgressFill');
  const progressText = document.getElementById('nerProgressText');
  const progressContainer = document.getElementById('nerDownloadProgress');
  const buttons = document.querySelectorAll('#nerDownloadModal .modal-footer button');

  // Show progress bar, hide buttons
  progressContainer.style.display = 'block';
  buttons.forEach(btn => btn.style.display = 'none');

  try {
    // Request background script to download/initialize NER
    const response = await chrome.runtime.sendMessage({
      type: 'INIT_NER',
      forceDownload: true
    });

    // Simulate progress (or get real progress from background)
    for (let i = 0; i <= 100; i += 10) {
      progressBar.style.width = `${i}%`;
      progressText.textContent = `Downloading... ${i}%`;
      await new Promise(resolve => setTimeout(resolve, 200));
    }

    if (response && response.success) {
      // Mark model as downloaded
      const settings = await chrome.storage.local.get(['settings']);
      settings.settings.nerModelDownloaded = true;
      settings.settings.nerEnabled = true;
      await chrome.storage.local.set({ settings: settings.settings });

      // Close modal, update UI
      document.getElementById('nerDownloadModal').style.display = 'none';
      document.getElementById('nerToggle').checked = true;
      updateStatusChip('ready');

      showToast('NER model downloaded successfully!');
    } else {
      throw new Error('Download failed');
    }
  } catch (error) {
    console.error('NER download error:', error);
    updateStatusChip('error');
    showToast('Failed to download NER model. Try again later.');

    // Reset modal
    progressContainer.style.display = 'none';
    buttons.forEach(btn => btn.style.display = '');
  }
}
```

#### 6. Enable/Disable NER Functions
```javascript
async function enableNER() {
  const settings = await chrome.storage.local.get(['settings']);
  settings.settings.nerEnabled = true;
  await chrome.storage.local.set({ settings: settings.settings });

  // Notify background script
  await chrome.runtime.sendMessage({ type: 'INIT_NER' });

  updateStatusChip('loading');
}

async function disableNER() {
  const settings = await chrome.storage.local.get(['settings']);
  settings.settings.nerEnabled = false;
  await chrome.storage.local.set({ settings: settings.settings });

  updateStatusChip('disabled');
}
```

#### 7. On Popup Load - Sync State
```javascript
document.addEventListener('DOMContentLoaded', async () => {
  const settings = await chrome.storage.local.get(['settings']);
  const currentSettings = settings.settings || {};

  // Set toggle state
  document.getElementById('nerToggle').checked = currentSettings.nerEnabled || false;

  // Set disabled if user chose "never ask"
  if (currentSettings.nerNeverAsk) {
    document.getElementById('nerToggle').disabled = true;
  }

  // Update status chip based on current state
  if (!currentSettings.nerEnabled) {
    updateStatusChip('disabled');
  } else if (!currentSettings.nerModelDownloaded) {
    updateStatusChip('error'); // Enabled but not downloaded = error state
  } else {
    // Check if NER is actually initialized
    const status = await chrome.runtime.sendMessage({ type: 'NER_STATUS' });
    if (status && status.initialized) {
      updateStatusChip('ready');
    } else {
      updateStatusChip('loading');
    }
  }
});
```

```javascript
function renderDetectionList(detections, element) {
  const listContainer = document.getElementById('pii-detections-list');
  listContainer.innerHTML = '';

  if (detections.length === 0) {
    listContainer.innerHTML = '<div class="no-detections">No PII detected</div>';
    return;
  }

  // Add detection summary
  const summary = createDetectionSummary(detections);
  listContainer.appendChild(summary);

  // Render each detection with source badge
  detections.forEach((detection, index) => {
    const item = document.createElement('div');
    item.className = 'detection-item';

    const source = detection.source || 'regex';
    const confidence = getConfidenceLevel(detection.confidence);

    item.innerHTML = `
      <div class="detection-header">
        <span class="detection-type">${formatType(detection.type)}</span>
        <div class="detection-badges">
          <span class="detection-source source-${source}">${source}</span>
          <span class="confidence-badge confidence-${confidence}">${confidence}</span>
        </div>
      </div>
      <div class="detection-value">${detection.value}</div>
      <div class="detection-actions">
        <button class="btn-mask" data-index="${index}">Mask</button>
        <button class="btn-remove" data-index="${index}">Remove</button>
      </div>
    `;

    listContainer.appendChild(item);
  });
}

function createDetectionSummary(detections) {
  const summary = document.createElement('div');
  summary.className = 'detection-summary';

  const regexCount = detections.filter(d => d.source === 'regex').length;
  const nerCount = detections.filter(d => d.source === 'ner').length;
  const total = detections.length;

  summary.innerHTML = `
    <div class="summary-title">Detected: ${total} items</div>
    <div class="detection-breakdown">
      <div class="breakdown-item">
        <span class="breakdown-count">${regexCount}</span> Regex
      </div>
      <div class="breakdown-item">
        <span class="breakdown-count">${nerCount}</span> NER
      </div>
    </div>
  `;

  return summary;
}

function getConfidenceLevel(confidence) {
  if (confidence >= 0.8) return 'high';
  if (confidence >= 0.6) return 'medium';
  return 'low';
}
```
